void print_carindb_map(){
    BigEndian(); 
    //#include "vdo_block.bt"
    
    local DWORD ptr, val, cnt_bl_types=0, cnt=1000000, cBlSize=0x800;
    local en_BL_DATA_TYPE type=_too_far, prev_type=_empty;
    
    
    //Printf ("%08X: %06X\t%08X\n", 0x800, ReadUInt(0x800) , (ReadUInt(0x800)>>8)*0x800 );
    cnt_bl_types = 1;
 //ru =    0x1Ea2d800
    
    //733470720 - EOF
    for(ptr=0; ptr< 0x1Ea2d800; ptr+=cBlSize){
    //for(ptr=0; ptr< ( cnt*cBlSize); ptr+=cBlSize){
        if(FSeek(ptr) != 0){
            Printf("seek\n"); break;}
        val = ReadUInt(ptr);
        if( ((val>>8)*cBlSize) == ptr){
            //Printf ("%08X: %08X\n", ptr, val);
            type = ReadByte(ptr+0x05);
            
            if( type != prev_type){
                prev_type = type;
                Printf (" --- %04i\n%08X: %08X :  %02x%s", 
                     cnt_bl_types,
                    ptr,   val,
                    type, EnumToString(type) );
                cnt_bl_types = 0;
             }
            cnt_bl_types ++;
        }
    }
    Printf (" --- %04i\n", cnt_bl_types);
}
